# Runtime dependencies only — this is what gets installed on the VM.
# Test-only packages live in requirements-dev.txt so the 12 GB box never
# installs pytest.
#
# ---------------------------------------------------------------------------
# torch: CPU build, pinned, on EVERY platform.
#
# PyPI's default torch for Linux drags in the entire CUDA toolkit — cudnn,
# cublas, nccl, triton, cusparselt and friends, several GB of it. Nothing in
# this project has an NVIDIA GPU: not the Oracle Ampere A1 VM, not the test
# container, not the GitHub Actions runners. It would all be dead weight on a
# box with 45 GB of disk.
#
# The +cpu build from PyTorch's own index avoids it, and 2.13.0 publishes
# wheels for manylinux_2_28_aarch64 (VM, container) and manylinux_2_28_x86_64
# (CI) too — so one pin covers every target and dev matches production
# exactly.
#
# This index is also the only source of a wheel matching the local
# development environment; PyPI has none at all for it.
# ---------------------------------------------------------------------------
--extra-index-url https://download.pytorch.org/whl/cpu

fastapi==0.115.6

# uvicorn, NOT uvicorn[standard]. The [standard] extra pulls in httptools,
# which has no wheel matching the local development environment, and uvloop,
# which does not support Windows. Splitting the dependency by platform would
# leave local and the VM running different server stacks — the exact
# divergence decision 23 removed by aligning Python to 3.12. The extra buys
# faster HTTP parsing and a faster event loop, neither of which is
# measurable here: every real request waits on a network call to the Claude
# API.
uvicorn==0.34.0

# Retrieval. torch is pinned explicitly so sentence-transformers cannot pull
# the CUDA build in as a transitive dependency.
torch==2.13.0+cpu
sentence-transformers==3.3.1

# sqlite-vec has no wheel matching the local development environment — those
# code paths are exercised in the Linux container (see backend/test/), on
# the VM and in CI, never natively here.
sqlite-vec==0.1.9

# Claude API client. Answer generation for /chat, and contact triage.
anthropic==0.119.0

# Backs pydantic's EmailStr on the contact form. The email address is how LJ
# replies to an enquiry, so a malformed one is a lost opportunity rather than a
# cosmetic problem — worth a real validator instead of a hopeful regex.
email-validator==2.3.0

# Reads backend/.env in local development. On the VM the same variables come
# from the systemd unit and in CI from GitHub Actions secrets, so this is inert
# there — but it is a runtime dependency, not a dev one, because config.py
# imports it unconditionally.
python-dotenv==1.2.2

# Error tracking. Off unless SENTRY_DSN is set, so it does nothing locally or in
# tests — but it is imported at startup on the VM, so it belongs in the runtime
# set, not the dev one.
sentry-sdk==2.66.1
